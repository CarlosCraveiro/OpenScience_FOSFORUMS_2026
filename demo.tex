\begin{frame}{What is Open Science?}
  \begin{block}{Definition (FAPESP / UNESCO)}
    A set of policies and actions to disseminate knowledge, making all research results \textbf{accessible}, \textbf{reusable}, and \textbf{reproducible}.
  \end{block}

  \vspace{1em}
  \begin{columns}[T,onlytextwidth]
    \column{.48\textwidth}
      \cookiekicker{The 3 Core Elements}
      \begin{itemize}
          \item \textbf{Open Access:} Publications \& articles.
          \item \textbf{Open Data:} Raw data, models, specs.
          \item \textbf{Open Source/Processes:} Software, algorithms, and methodologies.
      \end{itemize}

    \column{.48\textwidth}
      \cookiekicker{The Main Goal}
      \vspace{.5em}
      
      Accelerate scientific, technological, economic, and social progress through active collaboration and knowledge reuse.
  \end{columns}
\end{frame}

\begin{frame}{Corollaries: The Consequences of Open Science}
  \begin{itemize}
    \item \textbf{Efficiency in Public Spending}
      \begin{itemize}
        \item Reduces redundant data production.
        \item \textit{Examples:} Human Genome Project, Data Amazon (USP).
      \end{itemize}
      
    \vspace{0.5em}
    \item \textbf{Lowering Barriers to Research}
      \begin{itemize}
        \item Financial relief via data/infrastructure reuse.
        \item Better documentation enables cross-disciplinary collaboration.
      \end{itemize}
      
    \vspace{0.5em}
    \item \textbf{Boosting Industry Innovation}
      \begin{itemize}
        \item Clearer, replicable knowledge directly fuels market applications.
      \end{itemize}
      
    \vspace{0.5em}
    \item \textbf{Empowering Science Communication}
      \begin{itemize}
        \item Easier access for communicators to raise societal awareness.
      \end{itemize}
  \end{itemize}
\end{frame}
